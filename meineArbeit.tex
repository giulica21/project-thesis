\section{Introduction}
The Anderson localization model was introduced by the physicist P.W. Anderson in 1958 to
explain the quantum mechanical effects of disorder in materials. Anderson
discusses the behaviour of electrons in dirty crystals, which is the quantum mechanical analougue of a random walk in a random environment.
The model considers a d-dimensional grid, in which each site represents an atom. From an heuristical point of view, we also consider electrons
that jump from atom to atom and are subject to an external random potential, which influences their
motion in the material.
When these particles cannot jump , it indicates the absence of
quantum transport.This phenomenon is called Anderson localisation. This model deals with conditions for which the electrons
stay trapped in a specific region and don’t generate electric conduction, defining an insulator.  This is
in sharp contrast to the behaviour in ideal crystals, which are always conductors. \\
Anderson's work, and that of N.F.Mott and J.H. van Vleck, won the 1977 physics Nobel prize for "their fundamental theoretical
investigations of the electronic structure of magnetic and disordered systems". Since then, the study of random operators has become an important field of 
mathematical physics, which has led to a tremendous amount of research activity and many mathematical results. 
\\
The two important methods to prove Anderson localization are the \textit{multiscale analysis method} (MSA), developed by Froehlich and Spencer [] in 1983, and the 
\textit{ fractional moment method} (FMM) by Aizenman and Molchanov in 1993. We will deal with the second one, which is more elementary 
and gives stronger results on \textit{dynamical localization} [Chapter 2.2]. \\

\subsection{The Anderson Model}
Transport phenomena in a disordered material are often described via discrete random Schroedinger operator. On the lattice $\mathbb{Z}^d$, it takes the form of an infinite random matrix called the \textit{Hamiltonian}: 
\[
H_\omega = - \Delta + \lambda V_\omega.
\]
where $ -\Delta$ is a deterministic matrix and $ V_\omega$ is a diagonal matrix with random entries. This operator acts on the Hilbert space 
\[
l^2(\mathbb{Z}^d) = \{ u : \mathbb{Z}^d \to \mathbb{C} : \sum_{n \in \mathbb{Z}^d} |u(n)|^2 < \infty\}
\]
with inner product $\langle u, v \rangle = \sum_n \overline{u(n)} v(n)$. 
\\
\vspace{0.3 cm}\\
\subsubsection{The operator}
\textbf{The Descrete Laplacian.} The kinetic term $- \Delta$ is the descrete analogue of the usual negative Lalacian ($ -\sum_j \partial^2/\partial x_j^2$).
For the functions $u \in l^2(\mathbb{Z}^d)$ it is defined as:

\[
-\Delta u(x) := \sum_{k \in \mathbb{Z}^d,|x-k|=1} u(x) - u(k) \quad \text{for } x \in \mathbb{Z}^d.
\]

where $|*|$ denotes the $ l^2$ norm.

The discrete Laplacian represents the kinetic term and connects neighbour sites on the lattice, such that the electron can transition from one to another. 

For each site $x$, there are $2d$ possible $k$ such that $|x-k|=1$. We can now define an infinte matrix such that: \[
(-\Delta)_{x,k} = \begin{cases} 
2d & \text{if } x = k \quad \text{(on the diagonal)} \\
-1 & \text{if } |x - k| = 1 \quad (k \sim x, \text{ first neighbors}) \\
0 & \text{otherwise}
\end{cases}
\]



\vspace{1em}
\noindent
For $u \in \ell^2(\mathbb{Z}^d)$, $-\Delta u$ is summable in $\ell^2$, bounded, and self-adjoint.

The first claim (summability) is shown by:

\begin{align*}
\sum_{x \in \mathbb{Z}^d} |\Delta u(x)|^2 &= \sum_{x \in \mathbb{Z}^d} \left| \sum_{k \sim x, k \in \mathbb{Z}^d} \bigl(u(x) - u(k)\bigr) \right|^2 \\
&\leq 2d \sum_{x \in \mathbb{Z}^d} \left( \sum_{k \sim x} |u(x) - u(k)|^2 \right) \\
&\leq 2d \sum_{x \in \mathbb{Z}^d} \left( 2 \sum_{k \sim x, k \in \mathbb{Z}^d} |u(x)|^2 + |u(k)|^2 \right) \\
&\leq 2d \cdot 2d \cdot 4 \|u\|_2^2 < \infty.
\end{align*}

In each step we used, in order, Cauchy-Schwarz, the bound over the number of neighbors, triangle inequality and the invariance under translation of the $ l^2$ norm.\\
The boundedness comes from:
\[
\frac{\|\Delta u\|_2^2}{\|u\|_2^2} \le 16 d^2 \implies \|\Delta\|_2 \le 4d.
\]

To prove self-adjointness, we only need to show symmetry (given $-\Delta$ is bounded). For $ u, v \in l^2(\mathbb{Z}^d)$ and $ k \in \mathbb{Z}^d$:
\begin{align*}
\langle u, -\Delta v \rangle = \sum_{x \in \mathbb{Z}^d}\overline{u(x)} (-\Delta v(x)) &= \sum_{x \in \mathbb{Z}^d}u(x) \left( \sum_{k \sim x} v(x) - v(k) \right) \\
= \sum_{x \in \mathbb{Z}^d} \left(u(x) 2d v(x) -  \sum_{k \sim x} u(x) v(k)\right)  &=\sum_{x \in \mathbb{Z}^d} 2d u(x) v(x) - \sum_{x \in \mathbb{Z}^d} \sum_{k \sim x} u(k) v(x) \\
= \sum_{x \in \mathbb{Z}^d}v(x) \left( \sum_{k \sim x} u(x) - u(k) \right) &= \sum_{x \in \mathbb{Z}^d}\overline{v(x)} (-\Delta u(x)) =\langle - \Delta u, v \rangle.
\end{align*}

\noindent
The spectrum of the bounded, self-adjoint operator $-\Delta$ is the closed, real-valued interval $[0, 4d]$. In this range, we find the levels of kinetic energy that an electron owns.

\vspace{1em}
\noindent
In order to understand this results better, let's consider the discrete Fourier transform:
\[
\mathcal{F} \colon \ell^2(\mathbb{Z}^d) \longrightarrow L^2([0, 2\pi)^d)
\]
such that:
\[
(\mathcal{F}u)(x) \coloneqq \lim_{N \rightarrow \infty} (2\pi)^{-\frac{d}{2}} \sum_{|n| \leq N} u(n)\, e^{- i x \cdot n}, \qquad x \in [0, 2\pi)^d.
\]
For semplicity of the calculations we don't consider the limit and just sum over all $ n \in \mathbb{Z}^d$ and $ u$ summable. 

\begin{align*}
    (\mathcal{F} \Delta u)(x) &= (2\pi)^{-\frac{d}{2}} \sum_{n \in \mathbb{Z}^d} (\Delta u)(n)\, e^{-i x \cdot n} \\
    &= (2\pi)^{-\frac{d}{2}} \sum_{n \in \mathbb{Z}^d} \left( \sum_{|k-n|=1} \bigl(u(n) - u(k)\bigr) \right) e^{-i x \cdot n} \\
    &= (2\pi)^{-\frac{d}{2}} \sum_{n \in \mathbb{Z}^d} \left( 2d\, u(n) - \sum_{|k-n|=1} u(k) \right) e^{-i x \cdot n} \\
    &= 2d\,(\mathcal{F}u)(x) - (2\pi)^{-\frac{d}{2}} \sum_{n \in \mathbb{Z}^d} \sum_{|k-n|=1} u(k)\, e^{-i x \cdot n} \\
    &= 2d\,(\mathcal{F}u)(x) - (2\pi)^{-\frac{d}{2}} \sum_{k \in \mathbb{Z}^d} u(k) \sum_{|m|=1} e^{-i x \cdot (k+m)} \quad \text{\small (for $m = n - k$)} \\
    &= 2d\,(\mathcal{F}u)(x) - \left( (2\pi)^{-\frac{d}{2}} \sum_{k \in \mathbb{Z}^d} u(k)\, e^{-i x \cdot k} \right) \sum_{|m|=1} e^{-i x \cdot m} \\
    &= 2d\,(\mathcal{F}u)(x) - (\mathcal{F}u)(x) \sum_{j=1}^d \left( e^{-i x_j} + e^{i x_j} \right) \\
    &= \left( 2d - 2\sum_{j=1}^d \cos(x_j) \right) (\mathcal{F}u)(x)\\
    &= \left( 2 \sum_{j=1}^d 1-\cos(x_j) \right)  (\mathcal{F}u)(x).
\end{align*}
We obtain the multiplication operator: $$ \mathcal{F} \Delta \mathcal{F}^{-1}=  2 \sum_{j=1}^d 1-\cos(x_j) $$ on $ L^2 ( [0,2\pi)^d) $ in the variable $x= (x_1,\dots.., x_d)$. We notice that the Laplacian doesn't have eigenvalues in $ l^2(\mathbb{Z}^d)$.

\[
\phi_x(n) := e^{-i n \cdot x}
\]
While $\phi_x \notin l^2(\mathbb{Z}^d)$, $\Delta$ acts on it as
\begin{align*}
(\Delta \phi_x)(n) &= \sum_{|k|=1} e^{-in \cdot x} - e^{-i(n+k)\cdot x} \\

L'interpretazione fisica di questi sottospazi \`e chiarita dal fondamentale \textbf{Teorema RAGE} (Ruelle-Amrein-Georgescu-Enss):
\begin{theorem}[RAGE]
Suppose $H$ is a Schr\"odinger operator acting on $l^2(\mathbb{Z}^d)$. Then:
\begin{enumerate}
    \item[\textup{(i)}] $\phi \in \mathcal{H}_{pp}$ if and only if
    \[
    \lim_{R \to \infty} \sup_{t \in \mathbb{R}} \|\chi_{(|x| > R)} e^{-itH} \phi\| = 0.
    \]
    \item[\textup{(ii)}] $\psi \in \mathcal{H}_c = \mathcal{H}_{ac} \oplus \mathcal{H}_{sc}$ if and only if
    \[
    \lim_{T \to \infty} \frac{1}{2T} \int_{-T}^T \|\chi_{(|x| \le R)} e^{-itH} \psi\|\, dt = 0
    \]
    for all $R > 0$.
\end{enumerate}
Here $\chi_{(|x| \le R)}$ and $\chi_{(|x| > R)}$ denote the characteristic functions of the ball $\{|x| \le R\}$ and the complement of the ball, respectively.
\end{theorem}

\section{Localisation properties}
We understood what localisation means from an heuristical point of view. Mathematically, it can be described
from spectral and dynamical properties. 
\subsection{spectral localisation}


\begin{definition}[Spectral Localization]
 random Schr\"odinger operator $H = H_\omega$ exhibits \textbf{spectral localization} if $H_\omega$ almost surely has only pure point spectrum, that is,
\[
\sigma(H_\omega) = \sigma_{pp}(H_\omega) \quad \text{and} \quad \sigma_c(H_\omega) = \emptyset \quad \text{almost surely.}
\]
Furthermore the eigenfunctions corresponding to all eigenvalues decay exponentially: for almost all $\omega$, $H_\omega$ has a complete set of eigenfunctions $(\phi_{\omega,n})_{n \in \mathbb{N}}$ satisfying
\[
|\phi_{\omega,n}(x)| \le C_{\omega,n} e^{-\mu |x - x_{n,\omega}|}
\]
where $\mu > 0$ is the localization length parameter, $x_{n,\omega}$ are the centers of localization, and $C_{\omega,n} < \infty$ is a finite constant.
\end{definition}

\subsubsection{Absence of transport}

\begin{equation}
\langle x^2\rangle_{P_{H_\omega\in[a,b]}\phi_0}(t)
= \big\| \,|x|\, e^{-itH_\omega} P_{H_\omega\in[a,b]}\phi_0 \big\|^2
\tag{5}
\end{equation}

If the electrons with energies in $[a,b]$ move ballistically with average
velocity $v_{av}$, then roughly $x(t) \sim v_{av} t$ for large times. Hence
$\langle x^2\rangle_{P_{H_\omega\in[a,b]}\phi_0}(t)$ will be proportional to
$t^2$ for large $t$.

On the other hand, if the electrons in the energy interval $[a,b]$ are
localized, then it should be natural to expect that
$\langle x^2\rangle_{P_{H_\omega\in[a,b]}\phi_0}(t)$ is bounded uniformly in
$t$. It is known (Simon, 1990) that pure point spectrum implies absence of
ballistic motion,
\[
\lim_{t\to\infty} \frac{\langle x^2\rangle_{P_{H_\omega\in[a,b]}\phi_0}(t)}{t^2} = 0
\]
for all compactly supported initial conditions $\phi_0$, as soon as the
spectrum in $[a,b]$ is pure point. However, del Rio et al. (1996)
constructed examples of (non-random) one-dimensional Schr\"odinger operators
with pure point spectrum for all energies and exponentially localized
eigenfunctions for which
\begin{equation}
\limsup_{t\to\pm\infty} \frac{\langle x^2\rangle_{\phi_0}(t)}{t^{2-\delta}} = \infty
\qquad \text{for all } \delta > 0,
\tag{6}
\end{equation}
for a large class of compactly supported initial conditions $\phi_0$.
\textit{Il fallimento risiede nella libert\`a delle costanti $C_{\omega,n}$: se $C_{\omega,n}$ cresce incontrollatamente con $n$, le code delle autofunzioni possono sovrapporsi su scale spaziali arbitrariamente grandi.}



\subsection{dynamical localisation}

\begin{definition}[Dynamical localisation]
    \begin{equation}
    \sup_{t \in \mathbb{R}} \||X|^p e^{-it h_\omega}  \psi\| < \infty \quad \text{\emph{almost surely,}}
    \end{equation}
    \emph{where the position operator $|X|$ is defined by $(|X|\phi)(n) = |n|\phi(n)$ and for all $\varphi$ with compact support..}
\end{definition}

\begin{definition}[Exponential localisation]
    \begin{equation}
        \mathbb{E}\left( \sup_{t \in \mathbb{R}} |\langle e_j, e^{-itH_\omega}e_k \rangle | \right) \leq C e^{- \mu |j-k|}
    \end{equation}
\end{definition}



\textbf{Proposition:} \emph{Exponential dynamical localization implies that all moments of the position operator are bounded in time, i.e., dynamical localisation}


\begin{proof}
To see how (1) follows from exp. dynamical localization, assume that $\psi(k) = 0$ for $|k| > R$. Then
\begin{align*}
\||X|^p e^{-it h_\omega} \psi\|^2 &= \sum_j |\langle e_j, |X|^p e^{-it h_\omega}  \psi \rangle|^2 \\
&= \sum_j |j|^{2p} \left| \sum_{|k| \le R} \langle e_j, e^{-it h_\omega}  e_k \rangle \psi(k) \right|^2 \\
&\le \sum_j \sum_{|k| \le R} |j|^{2p} |\langle e_j, e^{-it h_\omega}  e_k \rangle|^2 \|\psi\|^2,
\end{align*}
where the last step used the Cauchy-Schwarz inequality. We can drop the square from $|\langle e_j, e^{-it h_\omega}  e_k \rangle|^2$ (as this number is bounded by $1$) and then take expectations to get
\begin{align*}
\mathbb{E}\left( \sup_t \||X|^p e^{-it h_\omega} \psi\|^2 \right) &\le \sum_j \sum_{|k| \le R} |j|^{2p} \mathbb{E}\left( \sup_t |\langle e_j, e^{-it h_\omega}  e_k \rangle| \right) \|\psi\|^2 \\
&\le C \sum_j \sum_{|k| \le R} |j|^{2p} e^{-\mu |j-k|} \|\psi\|^2 \\
&< \infty.
\end{align*}

\end{proof}


\subsection{From dynamical to spectral localisation}

\begin{proposition}
Suppose that dynamical localization in the form
\begin{equation}
\mathbb{E}\left(\sup_{t\in\mathbb{R}} |\langle e_j, e^{-it h_\omega} \chi_I(h_\omega) e_k \rangle|\right) \le C e^{-\mu |j-k|}
\end{equation}
holds in an open interval $I$. Then $h_\omega$ almost surely has pure point spectrum in $I$.
\end{proposition}

\begin{proof}
For a discrete Schr\"odinger operator $h = h_0 + V$ in $l^2(\mathbb{Z}^d)$ let $P_{\mathrm{cont}}(h)$ be the projection onto its continuous spectral subspace. Then the RAGE-Theorem says that for every $\psi \in l^2(\mathbb{Z}^d)$,
\begin{equation}
\|P_{\mathrm{cont}}(h) \chi_I(h) \psi\|^2 = \lim_{R\to\infty} \lim_{T\to\infty} \int_0^T \frac{dt}{T} \|\chi_{\{|x|\ge R\}} e^{-ith} \chi_I(h) \psi\|^2.
\end{equation}
If $\psi$ has finite support, say $\mathrm{supp}\,\psi \subset \{|x| \le r\}$, then
\begin{align*}
\|\chi_{\{|x|\ge R\}} e^{-ith} \chi_I(h) \psi\|^2 &\le \|\chi_{\{|x|\ge R\}} e^{-ith} \chi_I(h) \chi_{\{|x|\le r\}}\|^2 \|\psi\|^2 \\
&\le \|\chi_{\{|x|\ge R\}} e^{-ith} \chi_I(h) \chi_{\{|x|\le r\}}\| \|\psi\|^2 \\
&\le \sum_{|x|\ge R, |y|\le r} |\langle e_x, e^{-ith} \chi_I(h) e_y \rangle| \|\psi\|^2,
\end{align*}
where dropping a square is allowed as $\|\chi_{\{|x|\ge R\}} e^{-ith} \chi_I(h) \chi_{\{|x|\le r\}}\| \le 1$.

Taking expectations in (2) implies, after using Fatou and Fubini,
\begin{equation}
\mathbb{E}\left( \|P_{\mathrm{cont}}(h_\omega) \chi_I(h_\omega) \psi\|^2 \right) \le \lim_{R\to\infty, T\to\infty} \int_0^T \frac{dt}{T} \sum_{|x|\ge R, |y|\le r} \mathbb{E}\left( |\langle e_x, e^{-ith_\omega} \chi_I(h_\omega) e_y \rangle| \right) \|\psi\|^2.
\end{equation}
By (2.4) we have $\mathbb{E}\left( |\langle e_x, e^{-ith_\omega} \chi_I(h_\omega) e_y \rangle| \right) \le C e^{-\mu |x-y|}$ uniformly in $t$, which bounds the right hand side of (2.6) by
\[
\le \lim_{R\to\infty} \tilde{C} \sum_{|x|\ge R, |y|\le r} e^{-\mu |x-y|} = 0.
\]
We conclude that $P_{\mathrm{cont}}(h_\omega) \chi_I(h_\omega) \psi = 0$ for almost every $\omega$ and every $\psi$ of finite support. The latter are dense in $l^2(\mathbb{Z}^d)$ and thus $P_{\mathrm{cont}}(h_\omega) \chi_I(h_\omega) = 0$ almost surely, meaning that the spectrum in $I$ is pure point.
\end{proof}

\begin{proof}[Proof of a priori bound]
This proof demonstrates the simplicity of the fundamental idea underlying the FMM. For fixed $x \in \mathbb{Z}^d$, write $\omega = (\hat{\omega}, \omega_x)$ where $\hat{\omega}$ is short for $(\omega_u)_{u \in \mathbb{Z}^d \setminus \{x\}}$. With $P_{e_x} := \langle e_x, \cdot \rangle e_x$, the orthogonal projection onto the span of $e_x$, we can separate the $\omega_x$ and $\hat{\omega}$ dependence of $H_{\omega}$ as

For the operators $ (H_\omega-z)$ and $(H_{\hat{\omega}}-z)$, the resolvent identity yields
\begin{equation}
(H_{\omega} - z)^{-1} = (H_{\hat{\omega}} - z)^{-1} -  \lambda \omega_x ( H_{\hat{\omega}} - z)^{-1} P_{e_x} (H_{\omega} - z)^{-1}.
\end{equation}
Taking matrix-elements we conclude for the corresponding diagonal Green functions that
\begin{equation}
G_{\omega,\lambda}(x, x; z) = G_{\hat{\omega},\lambda}(x, x; z) - \lambda \omega_x G_{\hat{\omega},\lambda}(x, x; z) G_{\omega,\lambda}(x, x; z)
\end{equation}
and
\begin{equation}
G_{\omega,\lambda}(x, x; z) = \frac{1}{a + \lambda \omega_x} \quad \text{with } a = \frac{1}{G_{\hat{\omega},\lambda}(x, x; z)}.
\end{equation}
Note that the latter is well-defined since one can easily check that\\
 $\operatorname{Im} G_{\hat{\omega},\lambda}(x, x; z) >0 $ for $ \operatorname{Im} z > 0$.

The important fact is that $a$ is a complex number which does not depend on $\omega_x$. Thus, writing $\mathbb{E}_x(\dots) := \int \dots \rho(\omega_x)\, d\omega_x$ for $ \rho(\omega_x) = \frac{1}{2} \chi_{[-1,1]}$, we find that
\begin{equation}
\mathbb{E}_x(|G_{\omega,\lambda}(x, x; z)|^s) = \frac{1}{2 \lambda^s} \int_{-1}^{1} \frac{d\omega_x}{|\frac{a}{\lambda} + \omega_x|^s} \le \frac{1}{2 \lambda^s} \int_{-1}^{1} \frac{d\omega_x}{|\omega_x|^s} = \frac{1}{(1-s)\lambda^s},
\end{equation}
with $\frac{1}{1-s}$ independent of $\lambda$ and $a$, and thus independent of $\hat{\omega}$, $z$ and $x$.

\end{proof}


\subsubsection{From fractional moment's exponential bound to dynamical localisation}

In this subsection we aim to prove the following theorem:

\begin{theorem}
    Let $I \subset \mathbb{R}$ be an open bounded interval. If there exist $s \in (0,1)$, $C < \infty$ and $\mu > 0$ such that
    \begin{equation}
    \mathbb{E}\left( |G(x,y; E+i\epsilon)|^s \right) \le C e^{-\mu |x-y|}
    \end{equation}
    uniformly in $E \in I$ and $\epsilon > 0$, then dynamical localization in the form
    \begin{equation}
    \mathbb{E}\left( \sup_{t \in \mathbb{R}} |\langle e_j, e^{-it h_\omega} \chi_I(h_\omega) e_k \rangle| \right) \le C e^{-\mu |j-k|}
    \end{equation}
    holds on the interval $I$.
\end{theorem}
Before diving in the proof, we are going to need more equipment. 

\begin{lemma}
    For a bounded and self-adjoint operator A,
    \begin{equation}
        \left\| (A-z)^{-1} \right\| \le \frac{1}{\operatorname{Im}(z)}
    \end{equation}
    for all $ z \in \mathbb{C} \setminus \mathbb{R}$.
\end{lemma}
\begin{proof}
    For the definition of the operator norm, we know that 
    \[
    \left\|(A-z)^{-1}\right\| \coloneq \sup_{ \mu \in \sigma((A-z)^{-1})} |\mu|
    \]
    Let's consider a $ \lambda \in \mathbb{C}$ and write $ \mu = \frac{1}{\lambda -z}$. This means:
    \begin{align*}
    \frac{1}{\lambda-z} \in \sigma((A-z)^{-1}) &\iff (A-z)^{-1} - (\lambda-z)^{-1} \text{ not invertible} \\
    &\iff (A-z)^{-1} (A-\lambda)(\lambda-z)^{-1} \text{ not invertible} \\
    &\iff (A- \lambda) \text{ not invertible} \\
    &\iff \lambda \in \sigma(A)
    \end{align*}
    Where the resolvent identity has been used in third Implication.\\
    At this point we obtain 
    \[
    \left\|(A-z)^{-1}\right\| = \sup_{ \lambda \in \sigma(A)} \frac{1}{|\lambda -z|} = \frac{1}{\inf_{\lambda \in \sigma(A)} |\lambda-z|} = \frac{1}{\text{dist}(\sigma(A),z)}
    \]
    For a bounded and self-adjoint operator, we know that the spectrum is real-valued. For $ z \in \mathbb{C} \setminus \mathbb{R} $ the minimum distance between $ z$ and $ \sigma(A)$ is the imaginary component of $z$:\\
    \[
    \left\|(A-z)^{-1}\right\| \le \frac{1}{|\text{Im}(z)|}
    \]
    
\end{proof}

Thid bound is one of the most important and useful in this thesis. Once again we applied the resolvent identity to find a simple yet useful result. 


\begin{proposition}
For every $s \in (0,1)$ there exists a constant $C_1 < \infty$ only depending on $s$ and $\rho$ such that
\begin{equation}
|\mathrm{Im}\, z| \, \mathbb{E}_x\left(|G(x,y;z)|^2\right) \le C_1 \mathbb{E}_x\left(|G(x,y;z)|^s\right)
\end{equation}
for all $z \in \mathbb{C} \setminus \mathbb{R}$ and $x, y \in \mathbb{Z}^d$.
\end{proposition}

\begin{proof}
As in the proof of Lemma 3.2 we write $\omega = (\hat{\omega}, \omega_x)$. Keep $\hat{\omega}$ fixed and consider the Hamiltonian
\[
H^{(\alpha)} = H(\hat{\omega}, \omega_x + \alpha) = H_\omega + \alpha P_{e_x}.
\]
Its Green function will be denoted by $G^{(\alpha)}$. Applying the resolvent identity, we find
\[
(H_\omega - z)^{-1} = (H^{(\alpha)} - z)^{-1} + \alpha (H_\omega - z)^{-1} P_{e_x} (H^{(\alpha)} - z)^{-1}
\]
and

\begin{equation}
    \begin{split}
        G^{(\alpha)}(x,y;z) &= \frac{G_\omega(x,y;z)}{1 + \alpha G_\omega(x,x;z)} \\
        &= \frac{1}{\alpha + G_\omega(x,x;z)^{-1}} \cdot \frac{G_\omega(x,y;z)}{G_\omega(x,x;z)}.   
    \end{split}
\end{equation}



For the special case $x = y$ and $\tilde{\alpha} = -\mathrm{Re}\, G_\omega(x,x;z)^{-1}$ we get from (3.11) and Lemma 3.5 that
\[
\left|\frac{1}{\mathrm{Im}\, G(x,x;z)^{-1}}\right| = |G^{(\tilde{\alpha})}(x,x;z)| \le \frac{1}{|\mathrm{Im}\, z|},
\]
i.e.\ $|\mathrm{Im}\, G(x,x;z)^{-1}| \ge |\mathrm{Im}\, z|$. Inserting this into (3.11) gives
\begin{equation}
|\mathrm{Im}\, z| \, |G^{(\alpha)}(x,y;z)|^2 \le \frac{|\mathrm{Im}\, G_\omega(x,x;z)^{-1}|}{|\alpha + G_\omega(x,x;z)^{-1}|^2} \cdot \frac{|G_\omega(x,y;z)|^2}{|G_\omega(x,x;z)|^2}.
\end{equation}
On the other hand, we can bound the same expression by


\begin{align}
|\mathrm{Im}\, z| \, |G^{(\alpha)}(x,y;z)|^2 &\le |\mathrm{Im}\, z| \sum_{y' \in \mathbb{Z}^d} |G^{(\alpha)}(x,y';z)|^2 \nonumber \\
&= |\mathrm{Im}\, z| \langle e_x \, (H^{(\alpha)} - z)^{-1}, (H^{(\alpha)} - z)^{-1} \,e_x \rangle \nonumber \\
&= |\mathrm{Im}\, z| \langle e_x, (H^{(\alpha)} - \bar{z})^{-1} (H^{(\alpha)} - z)^{-1} e_x \rangle  \nonumber \\
&= |\mathrm{Im}\, z| \langle e_x, \frac{1}{z - \bar{z}} \left[ (H^{(\alpha)} - z)^{-1} - (H^{(\alpha)} - \bar{z})^{-1} \right] e_x \rangle \nonumber \\
&= |\mathrm{Im}\, G^{(\alpha)}(x,x;z)| \nonumber \\
&= \frac{|\mathrm{Im}\, G_\omega(x,x;z)^{-1}|}{|\alpha + G_\omega(x,x;z)^{-1}|^2},
\end{align}


where the third step used the symmetry of $ H^{(\alpha)}$ and last step used (3.11) with $x = y$.

For $t \ge 0$ one has $\min(1, t^2) \le t^s$, with $ s \in (0,1)$. Using this to interpolate between (3.12) and (3.13) we get
\begin{equation}
|\mathrm{Im}\, z| \, |G^{(\alpha)}(x,y;z)|^2 \le \frac{|\mathrm{Im}\, G_\omega(x,x;z)^{-1}|}{|\alpha + G_\omega(x,x;z)^{-1}|^2} \cdot \frac{|G_\omega(x,y;z)|^s}{|G_\omega(x,x;z)|^s}.
\end{equation}
We will now use the following ``re-sampling trick'', which has the effect of creating an additional random variable (here $\alpha$) to average over. For a non-negative Borel function $f$ on $\mathbb{R}$,
\begin{align}
\iint f(\omega_x + \alpha) \rho(\omega_x + \alpha)\, d\alpha\, \rho(\omega_x)\, d\omega_x &= \iint f(\omega_x + \alpha) \rho(\omega_x + \alpha) \rho(\omega_x)\, d\omega_x\, d\alpha \nonumber \\
&= \iint f(\omega_x) \rho(\omega_x) \rho(\omega_x - \alpha)\, d\omega_x\, d\alpha \nonumber \\
&= \int f(\omega_x) \rho(\omega_x) \left( \int \rho(\omega_x - \alpha)\, d\alpha \right) d\omega_x \nonumber \\
&= \int f(\omega_x) \rho(\omega_x)\, d\omega_x,
\end{align}
where in the first and third steps we used Fubini and in the second we used translation invariance of Lebesgue measure.

Choose $f(\omega_x) = |G_{(\hat{\omega}, \omega_x)}(x,y;z)|^2$; then (3.14) and (3.15) yield
\begin{align}
|\mathrm{Im}\, z| \, \mathbb{E}_x(|G_\omega(x,y;z)|^2) = |\mathrm{Im}\, z| \, \mathbb{E}_x\left( \int |G^{(\alpha)}(x,y;z)|^2 \rho(\omega_x + \alpha)\, d\alpha \right) \nonumber \\
\le \mathbb{E}_x\left( |\mathrm{Im}\, G_\omega(x,x;z)^{-1}| \frac{|G_\omega(x,y;z)|^s}{|G_\omega(x,x;z)|^s} \int \frac{\rho(\omega_x + \alpha)}{|\alpha + G_\omega(x,x;z)^{-1}|^2}\, d\alpha \right).
\end{align}
We now use Lemma 5.2 below with $w = G_\omega(x,x;z)^{-1}$ to conclude
\[
|\mathrm{Im}\, z| \, \mathbb{E}_x(|G_\omega(x,y;z)|^2) \le C \mathbb{E}_x(|G_\omega(x,y;z)|^s),
\]
with a constant $C < \infty$ which only depends on $\mathrm{supp}\,\rho$, but not on $x, y$ and $z$.
\end{proof}

\begin{lemma}
Let $0<s<1$. There exists a constant $C' = C'(\rho) < \infty$ such that
\[
|\mathrm{Im}\, w| \cdot |w|^s \int \frac{\rho(\omega_x + \alpha)}{|\alpha + w|^2}\, d\alpha \le C'
\]
uniformly in $w \in \mathbb{C}$ and $\omega_x \in \mathrm{supp}\,\rho$.
\end{lemma}

\begin{proof}
The proof is that of [REFERENCE GUNTER STOLZ]. Since $|w|^s \le |\alpha|^s + |\alpha + w|^s$, we bound:
\begin{enumerate}
    \item \begin{align*}
|\mathrm{Im}\, w| \int \frac{|\alpha|^s \rho(\omega_x + \alpha)}{|\alpha + w|^2}\, d\alpha &\le |\mathrm{Im}\, w| \int \frac{\| |\alpha|^s \rho(\omega_x + \alpha) \|_\infty}{|\alpha + w|^2}\, d\alpha \\
&\le |\mathrm{Im}\, w|\, \| |\alpha|^s \rho(\omega_x + \alpha) \|_\infty \int \frac{1}{(\alpha + u)^2 + t^2}\, d\alpha
\end{align*}
by setting $w = u + it$.
By changing variable and shift the integral, we have
\[
\int \frac{1}{\alpha^2 + t^2}\, d\alpha = 2\pi i \, \mathrm{Res}\left( \frac{1}{\alpha^2 + t^2}, it \right) = \frac{2\pi i}{2it}.
\]
Thus for $ \rho = \frac{1}{2}\chi_{[-1,1]}$ and $ |\alpha| \le 1+ |\omega| \le 2 $
\begin{align*}
|\mathrm{Im}\, w| \int \frac{|\alpha|^s \rho(\omega_x + \alpha)}{|\alpha + w|^2}\, d\alpha &\le \pi \| |\alpha|^s \rho(\omega_x + \alpha) \|_\infty \\
&\le \pi 2^{s-1}.
\end{align*}
    \item Now, we bound the second expression.
\begin{align*}
|\mathrm{Im}\, w| \int \frac{\rho(\omega_x + \alpha)}{|\alpha + w|^{2-s}}\, d\alpha &\le |\mathrm{Im}\, w|\, \frac{1}{2} \int \frac{d\alpha}{|\alpha + w|^{2-s}} \\
&= |t|\, \|\rho\|_\infty \int \frac{1}{|\alpha + it|^{2-s}}\, d\alpha \\
&= |t|^2 \|\rho\|_\infty \int \frac{1}{|t\beta + it|^{2-s}}\, d\beta \\
&= \|\rho\|_\infty t^s \int \frac{1}{|\beta + i|^{2-s}}\, d\beta
\end{align*}
by setting $w = u + it$ and $\alpha = t\beta$.

Let $C' = \int \frac{1}{|\beta + i|^{2-s}}\, d\beta$ then we have
\[
|\mathrm{Im}\, w| \int \frac{\rho(\omega_x + \alpha)}{|\alpha + w|^{2-s}}\, d\alpha \le \min\left( \frac{1}{|\mathrm{Im}\, w|^{1-s}}, C' \|\rho\|_\infty |\mathrm{Im}\, w|^s \right) \le C' \|\rho\|_\infty^{1-s}.
\] 
\end{enumerate}




\end{proof}
